function processDownload() { try { const urlParams = new URLSearchParams(window.location.search); const appKey = urlParams.get('app'); if (!appKey || !downloadDatabase.hasOwnProperty(appKey)) { window.location.href = "index.html"; return; } const targetLink = downloadDatabase[appKey]; if (targetLink === "COMING_SOON") { // Keeps your exact original delay and route-back logic untouched setTimeout(() => { window.location.href = "index.html"; }, 2000); } else { // Triggers the download silently in the background const silentDownloader = document.createElement('a'); silentDownloader.href = targetLink; silentDownloader.setAttribute('download', `${appKey}.apk`); silentDownloader.style.display = 'none'; document.body.appendChild(silentDownloader); silentDownloader.click(); document.body.removeChild(silentDownloader); // Routes back safely after the download starts without modifying any screen text setTimeout(() => { window.location.href = "index.html"; }, 3000); } } catch (err) { window.location.href = "index.html"; } }
